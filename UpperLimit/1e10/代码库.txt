def get_kde_smoothed_hist(df, score_col, weight_col, e_min, e_max, rue_cut, 
                          bins, bandwidth=0.15):
    """
    使用 KDE (核密度估计) 生成平滑的直方图 (类似 RooKeysPdf)。
    
    Args:
        bandwidth: 平滑系数 (bw_method)。
                   值越小(e.g. 0.05)越保留细节但可能有噪点；
                   值越大(e.g. 0.3)越平滑但可能抹平物理特征。
                   推荐 0.1 - 0.2。
    """
    # 1. 筛选数据 (获取 Raw Data)
    cond = (df['recE'] > e_min) & (df['recE'] < e_max) & \
           (df['rho40'] > CUT_RHO_MIN) & \
           (df['R_ue'] < rue_cut)
    
    df_cut = df[cond]
    
    # 2. 边界检查
    if len(df_cut) < 10: 
        # 如果统计量太少，KDE 会报错，回退到普通直方图或返回极小值
        print(f"[Warning] Not enough stats for KDE (N={len(df_cut)}). Using flat fallback.")
        return np.full(len(bins)-1, 1e-12)

    scores = df_cut[score_col].values
    weights = df_cut[weight_col].values
    
    # 3. 构建 KDE 模型
    try:
        kde = gaussian_kde(scores, weights=weights, bw_method=bandwidth)
    except Exception as e:
        print(f"[Error] KDE failed: {e}. Using flat fallback.")
        return np.full(len(bins)-1, 1e-12)
        
    # 4. 在 Bin 中心采样 (Resampling)
    bin_centers = (bins[:-1] + bins[1:]) / 2
    bin_widths = bins[1:] - bins[:-1]
    
    # kde(x) 返回概率密度 (pdf)，乘以宽度得到概率
    smoothed_probs = kde(bin_centers) * bin_widths
    
    # 5. 关键：重归一化 (Yield Restoration)
    # KDE 算出来的是归一化概率，必须乘回原始的总权重 (Yield)
    # 同时这步也修正了 KDE 在边界 [0,1] 外的泄漏问题
    total_yield = np.sum(weights)
    current_sum = np.sum(smoothed_probs)
    
    if current_sum > 0:
        smoothed_counts = smoothed_probs / current_sum * total_yield
    else:
        smoothed_counts = smoothed_probs
        
    return np.maximum(smoothed_counts, 1e-12)

def get_counts_varied(df, score_col, weight_col, e_min, e_max, rue_threshold=-3.0,  use_loose_template=False):
    """
    Args:
        rue_threshold: 想要测试的 Rue 阈值
        use_loose_template: 如果为 True，强制忽略传入的 rue_threshold，
                            改用 TEMPLATE_CUT (-1.5) 来获取高统计量的形状。
    """
    # 1. 基础物理 Cuts (能量 & rho40)
    cond = (df['recE'] > e_min) & (df['recE'] < e_max) & \
           (df['rho40'] > CUT_RHO_MIN)
    
    # 2. Rue Cut 逻辑 (核心修正)
    if use_loose_template:
        # 【背景专用】：强制使用宽松的模板阈值 (-1.5) 以保证统计量
        # 我们假设：能量标度改变带来的"形状变化趋势"，在 -1.5 和 -2.6 是一样的
        active_rue_cut = TEMPLATE_CUT
    else:
        # 【信号/数据专用】：使用真实的严格阈值 (-2.6 或变动值)
        active_rue_cut = rue_threshold

    cond = cond & (df['R_ue'] < active_rue_cut)
    df_cut = df[cond]
    
    # 3. 获取数据
    if len(df_cut) == 0:
        return np.full(len(BINS)-1, 1e-12) # 防止空数组报错

    scores = df_cut[score_col].values
    weights = df_cut[weight_col].values
    
    # 4. 生成直方图 (不归一化，只看相对变化)
    counts, _ = np.histogram(scores, bins=BINS, weights=weights,)
    
    return np.maximum(counts, 1e-12)

def get_normalized_shape(df, score_col, weight_col, e_min, e_max, rue_cut):
    """
    获取指定 Rue Cut 下的归一化形状 (Sum = 1)
    """
    cond = (df['recE'] > e_min) & (df['recE'] < e_max) & \
           (df['rho40'] > CUT_RHO_MIN) & \
           (df['R_ue'] < rue_cut) # 使用传入的测试 Cut
           
    df_cut = df[cond]
    
    if len(df_cut) == 0:
        return np.full(len(BINS)-1, 1e-12)

    scores = df_cut[score_col].values
    weights = df_cut[weight_col].values
    
    # 获取原始计数
    counts, _ = np.histogram(scores, bins=BINS, weights=weights)
    
    # *** 关键步骤：归一化 ***
    # 这消除了 Cut 变严导致的事例数减少，只保留“形状”变化
    integral = np.sum(counts)
    if integral > 0:
        normed_shape = counts  / integral
    else:
        normed_shape = counts
        
    return np.maximum(normed_shape, 1e-12)

def get_counts(df, score_col, weight_col, bins, scaled_num=100):
    """
    Helper to extract weighted histogram counts.
    Ensures no bin has 0 content (replaces with small epsilon) for fit stability.
    """
    scores = df[score_col].values
    weights = df[weight_col].values
    
    ratio_left = 1 - df[df[score_col] < bins[0]][weight_col].sum()/df[weight_col].sum()
    counts, gaps = np.histogram(scores, bins=bins, weights=weights, density=True)
    
    # Scale to annual yield if weights are rates (events/sec)
    # Assuming input weights are rates, multiply by seconds in a year
    # IF your weights are already event counts, remove this factor.
    # Based on previous context, weights seemed to be rates.
    SECONDS_PER_YEAR = 1 
    counts = counts * SECONDS_PER_YEAR * scaled_num * (bins[1] - bins[0]) * ratio_left
    
    # Avoid zero bins for statistical stability
    counts = np.maximum(counts, 1e-6)
    return counts
